# Hand Gesture Recognition System for CCTV Feeds
# Built by Akash Kumar | B.Tech CSE
import cv2
import numpy as np

def analyze_cctv_frame(frame):
    # 1. Grayscale & Gaussian Blur preprocessing
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    blurred = cv2.GaussianBlur(gray, (35, 35), 0)
    
    # 2. Otsu inverted binary thresholding
    _, thresh = cv2.threshold(blurred, 127, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
    
    # 3. Contour detection
    contours, _ = cv2.findContours(thresh.copy(), cv2.RETR_TREE, cv2.CHAIN_APPROX_NONE)
    
    if len(contours) > 0:
        max_c = max(contours, key=cv2.contourArea)
        hull = cv2.convexHull(max_c, returnPoints=False)
        defects = cv2.convexityDefects(max_c, hull)
        
        if defects is not None:
            finger_count = 0
            for i in range(defects.shape[0]):
                s, e, f, _ = defects[i, 0]
                start, end, far = tuple(max_c[s][0]), tuple(max_c[e][0]), tuple(max_c[f][0])
                
                # Cosine rule for angle calculation
                a = np.linalg.norm(np.array(end) - np.array(start))
                b = np.linalg.norm(np.array(far) - np.array(start))
                c = np.linalg.norm(np.array(end) - np.array(far))
                angle = np.arccos((b**2 + c**2 - a**2) / (2 * b * c))
                
                if angle <= np.pi / 2:  # Finger defect detected
                    finger_count += 1
            
            return finger_count + 1 # Classified gesture
    return 0
